  message: String,

internal data class ChoiceInspection(
  public val matchesAlpha: Boolean,
  public val matchesBeta: Boolean,
  public val failures: List<String>,
) {
  public val matchCount: Int
    get() = listOf(matchesAlpha, matchesBeta).count { it }
}

/**
 * Lossless anyOf wrapper for sdkgen://source/composition-golden.yaml#/components/schemas/Choice.
 */
@Serializable(with = Choice.Serializer::class)
public class Choice internal constructor(
  /**
   * Raw JSON retained as the serialization authority.
   */
  public val raw: JsonElement,
  private val json: Json,
  private val inspection: ChoiceInspection,
) {
  public val alpha: AlphaView? by
      lazy(LazyThreadSafetyMode.NONE) { if (inspection.matchesAlpha) json
        .decodeFromJsonElement<AlphaView>(raw) else null }

  public val beta: BetaView? by
      lazy(LazyThreadSafetyMode.NONE) { if (inspection.matchesBeta) json
        .decodeFromJsonElement<BetaView>(raw) else null }

  public val matchedBranches: Set<ChoiceBranch>
    get() = buildSet {
      if (inspection.matchesAlpha) add(ChoiceBranch.Alpha)
      if (inspection.matchesBeta) add(ChoiceBranch.Beta)
    }

  public companion object {
    /**
     * Builds a validated wrapper around raw JSON without rewriting it.
     */
    public fun fromRaw(raw: JsonElement, json: Json = SdkJson): Choice {
      val inspection = inspectChoice(raw)
      if (inspection.matchCount == 0) {
        throw ChoiceNoMatchException("Choice matched 0 branches: " + inspection.failures.joinToString("; "))
      }
      return Choice(raw, json, inspection)
    }
  }

  public object Serializer : KSerializer<Choice> {
    override val descriptor: SerialDescriptor = JsonElement.serializer().descriptor

    override fun deserialize(decoder: Decoder): Choice {
      val jsonDecoder = decoder.requireJsonDecoder("Choice")
      return fromRaw(jsonDecoder.decodeJsonElement(), jsonDecoder.json)
    }

    override fun serialize(encoder: Encoder, `value`: Choice) {
      encoder.requireJsonEncoder("Choice").encodeJsonElement(value.raw)
    }
  }
}

private fun inspectChoice(element: JsonElement): ChoiceInspection {
  val raw = element as? JsonObject ?: return ChoiceInspection(
    matchesAlpha = false,
    matchesBeta = false,
    failures = listOf("Alpha: expected JSON object", "Beta: expected JSON object"),
  )
  val matchesAlpha = raw["alpha"].isString()
  val matchesBeta = raw["beta"].isString()
  return ChoiceInspection(
    matchesAlpha = matchesAlpha,
    matchesBeta = matchesBeta,
    failures = buildList {
      if (!matchesAlpha) add("Alpha: required properties 'alpha' do not match their declared types")
      if (!matchesBeta) add("Beta: required properties 'beta' do not match their declared types")
    },
  )
}

/**
 * Closed oneOf union for sdkgen://source/composition-golden.yaml#/components/schemas/Variant.
 */
@Serializable(with = Variant.Serializer::class)
public sealed interface Variant {
  /**
   * Raw JSON retained as the serialization authority.
   */
  public val raw: JsonObject

  public class Cat internal constructor(
    public val cat: String,
    /**
     * Raw JSON retained as the serialization authority.
     */
    public override val raw: JsonObject,
  ) : Variant {
    public companion object {
      /**
       * Creates this branch and its canonical raw JSON representation.
       */
      public fun of(cat: String): Cat = Cat(
        cat = cat,
        raw = buildJsonObject {
          put("cat", cat)
        },
      )
    }
  }

  public class Dog internal constructor(
    public val dog: String,
    /**
     * Raw JSON retained as the serialization authority.
     */
    public override val raw: JsonObject,
  ) : Variant {
    public companion object {
      /**
       * Creates this branch and its canonical raw JSON representation.
       */
      public fun of(dog: String): Dog = Dog(
        dog = dog,
        raw = buildJsonObject {
          put("dog", dog)
        },
      )
    }
  }

  public object Serializer : KSerializer<Variant> {
    override val descriptor: SerialDescriptor = JsonElement.serializer().descriptor

    override fun deserialize(decoder: Decoder): Variant {
      val jsonDecoder = decoder.requireJsonDecoder("Variant")
      val raw = jsonDecoder.decodeJsonElement() as? JsonObject ?:
        throw VariantNoMatchException("Variant matched 0 branches: expected JSON object")
      val matches = inspectVariant(raw)
      if (matches.size == 0) {
        throw VariantNoMatchException("Variant matched 0 branches: " + matches.failures.joinToString("; "))
      }
      if (matches.size > 1) {
        throw VariantAmbiguityException("Variant matched " + matches.size + " branches; expected exactly 1: " + matches
          .names.joinToString())
      }
      return when {
        matches.catDecoded -> Cat(cat = requireNotNull(matches.cat), raw = raw)
        matches.dogDecoded -> Dog(dog = requireNotNull(matches.dog), raw = raw)
        else -> error("unreachable")
      }
    }

    override fun serialize(encoder: Encoder, `value`: Variant) {
      encoder.requireJsonEncoder("Variant").encodeJsonElement(value.raw)
    }
  }
}

private data class VariantInspection(
  public val cat: String?,
  public val catDecoded: Boolean,
  public val dog: String?,
  public val dogDecoded: Boolean,
  public val failures: List<String>,
) {
  public val names: List<String>
    get() = buildList {
      if (catDecoded) add("Cat")
      if (dogDecoded) add("Dog")
    }

  public val size: Int
    get() = names.size
}

private fun inspectVariant(raw: JsonObject): VariantInspection {
  val catResult = raw["cat"]?.let { element -> runCatching { SdkJson.decodeFromJsonElement<String>(element) } }
  val cat = catResult?.getOrNull()
  val catDecoded = catResult?.isSuccess == true
  val dogResult = raw["dog"]?.let { element -> runCatching { SdkJson.decodeFromJsonElement<String>(element) } }
  val dog = dogResult?.getOrNull()
  val dogDecoded = dogResult?.isSuccess == true
  return VariantInspection(
    cat = cat,
    catDecoded = catDecoded,
    dog = dog,
    dogDecoded = dogDecoded,
    failures = buildList {
      if (!catDecoded) add("Cat: required properties 'cat' do not match their declared types")
      if (!dogDecoded) add("Dog: required properties 'dog' do not match their declared types")
    },
  )
}

private fun JsonObject.stringValue(name: String): String? {
  val primitive = this[name] as? JsonPrimitive ?: return null
  return primitive.takeIf { it.isString }?.content
}
